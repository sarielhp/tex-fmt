\begin{algorithmic}[1]
\Procedure{Euclid}{$a, b$}
\State $r \gets a \bmod b$
\While{$r \neq 0$}
\State $a \gets b$
\State $b \gets r$
\State $r \gets a \bmod b$
\EndWhile
\State \Return $b$
\EndProcedure
\end{algorithmic}
